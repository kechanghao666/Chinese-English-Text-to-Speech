cmake_minimum_required(VERSION 3.16)
project(SpeechReader LANGUAGES CXX)
add_executable(SpeechReader WIN32 src/main.cpp)
target_compile_features(SpeechReader PRIVATE cxx_std_17)
target_link_libraries(SpeechReader PRIVATE sapi ole32 uuid comctl32 comdlg32 shell32)
if(MINGW)
  target_compile_options(SpeechReader PRIVATE -finput-charset=UTF-8)
  target_link_options(SpeechReader PRIVATE -municode -static)
endif()
if(MSVC)
  target_compile_options(SpeechReader PRIVATE /utf-8)
endif()
